    # Greedy algorithm
    max_iterations = 500  # Safety limit
    iteration = 0
    
    while iteration < max_iterations:
        iteration += 1
        
        # Check if all targets meet 90%
        all_met = True
        for county, specialty, total, threshold in targets:
            covered = len(coverage_sets[(county, specialty)])
            pct = covered / total
            if pct < threshold:
                all_met = False
                break
        
        if all_met:
            break
        
        # Find entity with maximum marginal gain
        best_entity = None
        best_gain = -1
        best_marginal = None
        
        for entity, cov in entity_coverage.items():
            if entity in [e for e in added_entities]:

            marginal = {}
            total_new = 0
            for key, new_members in cov.items():
                if key in coverage_sets:
                    already_covered = coverage_sets[key]
                    truly_new = new_members - already_covered
