    top_5_proximity = proximity_df.head(5).copy()
    
    # Create a balanced score between proximity and effectiveness
    # Normalize both metrics to 0-100 scale
    
    # For proximity: lower distance is better, so use inverse
    # Normalize distance: score = 100 * (max_dist - dist) / (max_dist - min_dist)
    max_dist = top_5_proximity["avg_distance_to_members_miles"].max()
    min_dist = top_5_proximity["avg_distance_to_members_miles"].min()
    
    # Avoid division by zero
    if max_dist == min_dist:
        top_5_proximity["proximity_score"] = 100
    else:
        top_5_proximity["proximity_score"] = 100 * (max_dist - top_5_proximity["avg_distance_to_members_miles"]) / (max_dist - min_dist)
    
    # For effectiveness: higher is better
    # Normalize effectiveness: score = 100 * (eff - min_eff) / (max_eff - min_eff)
    max_eff = top_5_proximity["avg_effectiveness"].max()
    min_eff = top_5_proximity["avg_effectiveness"].min()

    # Balanced score: equal weight (50% proximity, 50% effectiveness)
