    wayne_card_threshold = 15.0 / 69.0

    washtenaw_card_threshold = 15.0 / 69.0
    
    # Build BallTrees for members
    wayne_tree = BallTree(np.deg2rad(wayne_members), metric='haversine')
    washtenaw_tree = BallTree(np.deg2rad(washtenaw_members), metric='haversine')

    relevant_specialties = ["general practice", "cardiology"]
    relevant_candidates = candidates_df[candidates_df["specialty"].isin(relevant_specialties)]

    entity_providers = relevant_candidates.groupby("entity")

        provider_coords = np.deg2rad(providers[["lat", "lon"]].values)
