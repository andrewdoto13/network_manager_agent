    # Pareto-optimal analysis
    # A combination is Pareto-optimal if no other combination has both:
    # - Higher (or equal) coverage AND fewer (or equal) providers
    
    # Sort by coverage descending, then providers ascending
    combo_results.sort(key=lambda x: (-x["total_covered"], x["total_providers"]))
