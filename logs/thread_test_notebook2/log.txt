    # Calculate geographic proximity of each entity to members
    # Using haversine distance from entity's average location to member centroids
    
    from math import radians, sin, cos, sqrt, atan2
    
    def haversine(lat1, lon1, lat2, lon2):
        R = 3959  # Earth radius in miles
        lat1, lon1, lat2, lon2 = map(radians, [lat1, lon1, lat2, lon2])
        dlat = lat2 - lat1
        dlon = lon2 - lon1
        a = sin(dlat/2)**2 + cos(lat1) * cos(lat2) * sin(dlon/2)**2
        return R * 2 * atan2(sqrt(a), sqrt(1-a))
    
    # Get entity stats from cache
    entity_stats = pd.DataFrame(sandbox_cache.get("entity_stats"))
    member_centroids = pd.DataFrame(sandbox_cache.get("member_centroids"))
    
    # Calculate distance from each entity to each member centroid
    # We'll compute a weighted average distance based on member count
    entity_distances = []
    for _, entity_row in entity_stats.iterrows():
        entity_name = entity_row['entity']
        entity_lat = entity_row['avg_lat']
        entity_lon = entity_row['avg_lon']
        
        total_weighted_dist = 0
        total_members = 0
        for _, member_row in member_centroids.iterrows():
            dist = haversine(entity_lat, entity_lon, member_row['member_lat'], member_row['member_lon'])
            total_weighted_dist += dist * member_row['member_count']
            total_members += member_row['member_count']
        
        avg_distance = total_weighted_dist / total_members if total_members > 0 else float('inf')
        entity_distances.append({
            'entity': entity_name,
            'avg_distance_to_members': avg_distance,
            'provider_count': entity_row['provider_count'],
            'avg_effectiveness': entity_row['avg_effectiveness']
        })
    
    entity_dist_df = pd.DataFrame(entity_distances)
